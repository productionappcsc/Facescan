let busy = false; let recentScans = []; // เก็บ { descriptor, empId, time } ของคนที่เพิ่งสแกนแต่ละคน const COOLDOWN_MS = 5000; // เวลากันสแกนซ้ำต่อคน (5 วิ เท่าเดิม) const MATCH_THRESHOLD = 0.5; // ค่ายิ่งน้อยยิ่งเข้มงวดว่าเป็น "หน้าเดิม" // เช็คว่าหน้าที่จับได้ตอนนี้ ตรงกับคนที่เพิ่งสแกนไปเมื่อครู่หรือไม่ function findRecentScan(descriptor) { const now = Date.now(); recentScans = recentScans.filter(s => now - s.time < COOLDOWN_MS); // ล้างคนที่หมด cooldown แล้ว return recentScans.find(s => faceapi.euclideanDistance(s.descriptor, descriptor) < MATCH_THRESHOLD); } async function loop() { const options = new faceapi.TinyFaceDetectorOptions(); setInterval(async () => { if (busy) return; // ยังคงกันยิง request ซ้อนระหว่างรอผลลัพธ์ const detection = await faceapi.detectSingleFace(video, options) .withFaceLandmarks().withFaceDescriptor(); const ctx = overlay.getContext('2d'); ctx.clearRect(0, 0, overlay.width, overlay.height); if (!detection) return; faceapi.draw.drawDetections(overlay, [detection]); // ── ชั้นใหม่: เช็คก่อนว่าใบหน้านี้เพิ่งสแกนไปหรือยัง (แยกตามคน) ── const recent = findRecentScan(detection.descriptor); if (recent) { const secLeft = Math.ceil((COOLDOWN_MS - (Date.now() - recent.time)) / 1000); result.className = 'result idle'; result.textContent = `⏳ ${recent.empId} สแกนไปแล้ว รออีก ${secLeft} วิ`; return; } busy = true; result.className = 'result idle'; result.textContent = '⏳ กำลังตรวจสอบ...'; try { const res = await fetch(SCRIPT_URL, { method: "POST", headers: { 'Content-Type': 'text/plain' }, body: JSON.stringify({ action: 'scan', descriptor: Array.from(detection.descriptor) }) }); const data = await res.json(); if (data.status === 'success') { const label = data.type === 'IN' ? '✅ ลงเวลาเข้างาน' : '👋 ลงเวลาออกงาน'; result.className = 'result success'; result.textContent = `${label}\n${data.empName} (${data.empId})`; // ── บันทึก cooldown เฉพาะคนนี้เท่านั้น ไม่กระทบคนอื่น ── recentScans.push({ descriptor: detection.descriptor, empId: data.empId, time: Date.now() }); if (autoCloseAfterScan) { stopCameraAndClose(); } } else if (data.status === 'no_match') { result.className = 'result fail'; result.textContent = '❌ ไม่พบข้อมูลใบหน้านี้ในระบบ'; } else { result.className = 'result fail'; result.textContent = '⚠️ เกิดข้อผิดพลาด: ' + (data.message || 'ไม่ทราบสาเหตุ'); } } catch (err) { result.className = 'result fail'; result.textContent = '❌ เชื่อมต่อเซิร์ฟเวอร์ไม่ได้'; } setTimeout(() => { busy = false; result.className = 'result idle'; result.textContent = '🔍 กรุณาหันหน้าเข้ากล้อง'; }, 3000); }, 700); }
